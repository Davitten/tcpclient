find_package(Protobuf REQUIRED)
include_directories(${PROTOBUF_INCLUDE_DIR})
include_directories(${CMAKE_CURRENT_BINARY_DIR})
message(STATUS "Generating proto files")
file(GLOB PROTO_FILES "*.proto")
message(STATUS "Found following proto files: ${PROTO_FILES}")
protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS ${PROTO_FILES})
add_library(protos ${PROTO_SRCS})
target_link_libraries(protos PRIVATE ${PROTOBUF_LIBRARY})
message(STATUS "Generating proto files done")
